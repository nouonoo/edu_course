```mermaid
        +get_courses_for_user()

        +get_rating()
        +get_report_data()

        -_recalculate_assignment_score()

    class CourseProgress {
        <<module>>
        +PASS_THRESHOLD
        +calculate_section_score()

        +assignment_status_from_score()
        +is_course_passed()
    }

    class CourseManifest {
        <<module>>
        +load_course_manifest()
        +is_scorable_section()
    }

    class ScormRuntime {
        <<module>>
        +load_scorm_state()
        +save_scorm_state()
        +sync_scorm_progress()
    }

    class ScormImporter {
        <<module>>
        +import_course_archive()
    }

    FlaskApp --> Database : uses
    FlaskApp --> CourseProgress : uses
    FlaskApp --> CourseManifest : uses
    FlaskApp --> ScormRuntime : uses
    FlaskApp --> ScormImporter : uses
    Database --> CourseProgress : uses
    Database --> CourseManifest : uses
```

## Обоснование структуры классов

| Решение | Обоснование |
|---------|-------------|
| Один класс `Database` | Учебный проект; все SQL в одном месте, без ORM |
| Модули `course_progress`, `course_manifest` | Разделение чистой логики и I/O |
| `LMSBridge` отдельно от `CourseApp` | Плеер курса может работать автономно при тестировании |
| `CommonJS` как общий слой | DRY для auth и fetch на всех страницах LMS |
